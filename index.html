/* ========================================================= SANTA GLTF LOADER ========================================================= */ function loadSantaModel(){ modelNotice.style.display="block"; modelNotice.textContent = "🛷 Loading Santa's sleigh..."; console.log( "Starting Santa GLTF loader..." ); const modelURL = new URL( MODEL_FILE, window.location.href ).href; console.log( "Looking for:", modelURL ); if(typeof THREE === "undefined"){ console.error( "Three.js did not load." ); modelNotice.textContent = "⚠️ Three.js failed to load."; return; } if(typeof THREE.GLTFLoader === "undefined"){ console.error( "GLTFLoader did not load." ); modelNotice.textContent = "⚠️ GLTFLoader failed to load."; return; } const loader = new THREE.GLTFLoader(); loader.load( modelURL, function(gltf){ console.log( "🎅 santa.gltf successfully loaded!" ); santaModel = gltf.scene; santaModel.scale.set( .08, .08, .08 ); santaModel.visible = trackingStarted; santaModel.traverse( function(object){ if(object.isMesh){ object.frustumCulled=false; if(object.material){ object.material.needsUpdate=true; } } } ); try{ globe.scene().add( santaModel ); console.log( "🎅 Santa model added to globe scene." ); modelNotice.style.display = "none"; }catch(error){ console.error( "Could not add Santa to globe:", error ); modelNotice.textContent = "⚠️ Santa loaded but could not be placed on the globe."; } }, function(xhr){ if(xhr && xhr.total){ const percent = Math.round( xhr.loaded / xhr.total * 100 ); modelNotice.textContent = "🛷 Loading Santa's sleigh... " + percent + "%"; } }, function(error){ console.error( "❌ Santa GLTF loading error:", error ); modelNotice.textContent = "⚠️ santa.gltf could not be loaded."; } ); }
